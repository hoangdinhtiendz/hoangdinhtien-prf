ten = "Hoàng Đình Tiến"

bai2_tinh_tuoi.pyBài 2
from datetime import date

sinh = date(2007, 11, 6)
hom_nay = date.today()
tuoi = hom_nay.year - sinh.year - (
    (hom_nay.month, hom_nay.day) < (sinh.month, sinh.day)
)
print(f"Tuổi: {tuoi}")
bai3_dem_ngay_ra_quan.pyBài 3
from datetime import date

nhap_ngu = date(2026, 3, 5)
ra_quan = date(2028, 1, 10)
tong = (ra_quan - nhap_ngu).days
qua = (date.today() - nhap_ngu).days
print(f"Đã qua {qua}/{tong} ngày ({qua/tong:.2%})")
bai4_fizzbuzz.pyBài 4
for i in range(1, 16):
    if i % 15 == 0:
        print("FizzBuzz")
    elif i % 3 == 0:
        print("Fizz")
    elif i % 5 == 0:
        print("Buzz")
    else:
        print(i)

08Ước mơ & định hướng

    
“Có những ước mơ không ồn ào, chỉ lặng lẽ lớn lên trong từng bước chân hành quân, từng buổi sáng thức dậy trước tiếng kèn.”

    

      
      

        
Học viện Kỹ thuật Quân sự

        
Ước mơ lớn nhất: được khoác lên mình bộ quân phục của người học viên kỹ thuật quân sự, rèn tri thức và bản lĩnh, để một ngày dùng kỹ thuật và công nghệ góp phần xây dựng quân đội, bảo vệ Tổ quốc.

      

    

    

      
🎖️
Màu áo lính
Khoác lên mình bộ quân phục, mình học cách đứng thẳng giữa nắng gió, học kỷ luật và học hai chữ “đồng đội”. Hai năm quân ngũ là hai năm tôi luyện để trưởng thành.

      
📚
Cánh cửa giảng đường
Ngày ra quân không phải điểm dừng mà là một khúc rẽ mới. Mình sẽ cầm lại cây bút, thi lại Đại học với bản lĩnh vững vàng hơn rất nhiều.

      
🚀
Ngọn lửa công nghệ
Dù ở đâu, mình vẫn giữ lửa với Công nghệ thông tin: học, thử, sai rồi làm lại, để một ngày đủ bản lĩnh làm chủ kỹ thuật phục vụ quân đội.
